"""## 1. From a closed surface to a binary voxel array

An STL file stores a surface as triangles. It does not directly identify inside voxels. FakeCT:

1. creates a grid around the centered mesh;
2. computes the physical coordinate of each voxel center;
3. evaluates a mesh winding number at each center; and
4. labels `winding_number > 0.5` as inside.

The essential production idea is:

```python
WN = igl.fast_winding_number_for_meshes(vertices, faces, query_points)
inside = WN.reshape((N, N, N)) > 0.5
surface = binary_dilation(inside) ^ inside
outside = ~(inside | surface)
```

**Coordinate warning:** FakeCT volumes are stored in array order `(z, y, x)`, although geometric points are usually written `(x, y, z)`.

## 2. Begin in 2D: Boolean arrays and bitwise operations

A binary image is a set of `True` pixels.

| NumPy expression | Set idea | Meaning for masks |
|---|---|---|
| `A & B` | intersection | true in both |
| `A \| B` | union | true in either |
| `~A` | complement | not in A |
| `A ^ B` | symmetric difference | changed between A and B |

Use `&` and `|` for NumPy arrays. Python's words `and` and `or` do not perform element-by-element operations.
"""

"""### 2.1 Morphology is an operation across shifted copies

Four-neighbor dilation asks whether the center **OR** north, south, west, or east was `True`. Four-neighbor erosion asks whether the center **AND** all four neighbors were `True`.

```text
          north
            |
west --- center --- east
            |
          south
```

"""

def neighbors4(mask):
    '''Return north, south, west, and east arrays aligned with mask.'''
    north = np.pad(mask[:-1, :], ((1, 0), (0, 0)), constant_values=False)
    south = np.pad(mask[1:, :], ((0, 1), (0, 0)), constant_values=False)
    west = np.pad(mask[:, :-1], ((0, 0), (1, 0)), constant_values=False)
    east = np.pad(mask[:, 1:], ((0, 0), (0, 1)), constant_values=False)
    return north, south, west, east

def dilate4(mask):
    north, south, west, east = neighbors4(mask)
    return mask | north | south | west | east

def erode4(mask):
    north, south, west, east = neighbors4(mask)
    return mask & north & south & west & east

eroded_2d = erode4(vessel_2d)
dilated_2d = dilate4(vessel_2d)

fig, axes = plt.subplots(1, 3, figsize=(10, 3))

"""### 2.2 Constrain the operation to an ROI

Preserve the original outside the ROI and insert the candidate result only inside:

```python
result = (original & ~roi) | (candidate & roi)
```

This is the core connection between ROI selection and bitwise operations.

"""

def circular_roi_2d(shape_yx, center_yx, radius):
    y, x = np.indices(shape_yx)
    cy, cx = center_yx
    return (y - cy) ** 2 + (x - cx) ** 2 <= radius ** 2

def edit_binary_2d(mask, roi, operation, iterations=1):
    result = mask.copy()
    step = {"erosion": erode4, "dilation": dilate4}[operation]
    for _ in range(iterations):
        candidate = step(result)
        result = (result & ~roi) | (candidate & roi)
    return result

roi_2d = circular_roi_2d(vessel_2d.shape, center_yx=(7, 14), radius=5)
operation_iterations = 1 if ASSIGNED_TRACK == "erosion" else 2

    vessel_2d, roi_2d, ASSIGNED_TRACK, iterations=operation_iterations

"""### The two tracks are intentionally asymmetric

**Erosion / stenosis releases artery pixels**

```python
released = old_artery & ~new_artery
```

A recipient tissue must fill each released location. The algorithm needs neighbor evidence, tie-breaking, protected labels, and an unresolved output.

**Dilation borrows surrounding pixels**

```python
requested_borrow = new_artery & ~old_artery
```

Each requested location already has a donor label. The algorithm needs donor eligibility, protected labels, blocked requests, and per-label audit counts.

## 4D. Dilation track - first-pass 2D borrow algorithm

Complete this track only if `ASSIGNED_TRACK == "dilation"`.

For each requested borrow pixel:

1. read its current donor label;
2. borrow only if the donor appears in `allowed_donors` and is not protected;
3. set eligible locations to the artery label;
4. return a Boolean `blocked_mask` for denied requests;
5. return a dictionary counting borrowed pixels by original donor label; and
6. change nothing outside `requested_borrow_mask`.
"""

import numpy as np

def borrow_pixels_2d(grid, requested, artery_id, allowed_donors, protected_labels):
    updated = grid.copy()
    blocked = np.zeros_like(grid, dtype=bool)
    donor_counts = {}

    # 1. Protection > Permission
    valid_donors = set(allowed_donors) - set(protected_labels)

    # 2. Check if_artery
    is_already_artery = (grid == artery_id)

    # 3. Mask of borrowable pixels: requested AND valid_donor AND NOT protected
    borrowable_mask = requested & np.isin(grid, list(valid_donors)) & ~np.isin(grid, list(protected_labels))

    # 4. Blocked pixels: requested, NOT borrowable, AND NOT already the target artery
    blocked_mask = requested & ~borrowable_mask & ~is_already_artery
    blocked[blocked_mask] = True

    # 5. Apply updates to borrowable pixels
    borrowed_labels = grid[borrowable_mask]
    updated[borrowable_mask] = artery_id

    # 6. Tally donor counts for newly borrowed pixels only
    if borrowed_labels.size > 0:
        unique_labels, counts = np.unique(borrowed_labels, return_counts=True)
        for label, count in zip(unique_labels, counts):
            donor_counts[int(label)] = int(count)

    return updated, donor_counts, blocked

# borrow pixels
def borrow_pixels_2d(grid, requested, artery_id, allowed_donors, protected_labels):  #np_array of integers, true/false binary grid of pixel requests, id to match, what can be borrowed, what is protected
    updated = grid.copy() #create copy
    blocked = np.zeros_like(grid, dtype=bool) # initial unblocked mask
    donor_counts = {} # empty dict

    # 1. Protection > permission
    valid_donors = set(allowed_donors) - set(protected_labels) # valid donors set

    # 2. Identify pixels that are already the artery
    is_already_artery = (grid == artery_id)

    # 3. Borrowable: requested AND valid_donors AND NOT protected_labels
    borrowable_mask = ( # this is the bool mask which says which can and cannot be changed into artery
        requested #must be requested
        & np.isin(grid, list(valid_donors)) # must be in valid donors
        & ~np.isin(grid, list(protected_labels)) # must not be in protected labels
    )

    # 4. Blocked pixels: requested, NOT borrowable, AND NOT already the target artery
    blocked_mask = requested & ~borrowable_mask & ~is_already_artery #creating the rule
    blocked[blocked_mask] = True # applying the rule

    # 5. Apply updates
    borrowed_labels = grid[borrowable_mask] #grid of the borrowable label ids
    updated[borrowable_mask] = artery_id #updated

    # 6. Tally donor counts for newly borrowed pixels only
    if borrowed_labels.size > 0:
        unique_labels, counts = np.unique(borrowed_labels, return_counts=True) # 2 arrays: unique tissues and their counts after deduplicating and sorting
        for label, count in zip(unique_labels, counts): # populate donor counts dictionary after comparing the zip of the two arrays
            donor_counts[int(label)] = int(count)

    return updated, donor_counts, blocked

# borrow pixels
def borrow_pixels_2d(grid, requested, artery_id, allowed_donors, protected_labels):  #np_array of integers, true/false binary grid of pixel requests, id to match, what can be borrowed, what is protected
    updated = grid.copy() #create copy
    blocked = np.zeros_like(grid, dtype=bool) # initial unblocked mask
    donor_counts = {} # empty dict

    # 1. Protection > permission
    valid_donors = set(allowed_donors) - set(protected_labels) # valid donors set

    # 2. Identify pixels that are already the artery
    is_already_artery = (grid == artery_id)

    # 3. Borrowable: requested AND valid_donors AND NOT protected_labels
    borrowable_mask = ( # this is the bool mask which says which can and cannot be changed into artery
        requested #must be requested
        & np.isin(grid, list(valid_donors)) # must be in valid donors
        & ~np.isin(grid, list(protected_labels)) # must not be in protected labels
    )

    # 4. Blocked pixels: requested, NOT borrowable, AND NOT already the target artery
    blocked_mask = requested & ~borrowable_mask & ~is_already_artery #creating the rule
    blocked[blocked_mask] = True # applying the rule

    # 5. Apply updates
    borrowed_labels = grid[borrowable_mask] #grid of the borrowable label ids
    updated[borrowable_mask] = artery_id #updated

    # 6. Tally donor counts for newly borrowed pixels only
    if borrowed_labels.size > 0:
        unique_labels, counts = np.unique(borrowed_labels, return_counts=True) # 2 arrays: unique tissues and their counts after deduplicating and sorting
        for label, count in zip(unique_labels, counts): # populate donor counts dictionary after comparing the zip of the two arrays
            donor_counts[int(label)] = int(count)

    return updated, donor_counts, blocked

import numpy as np


def borrow_pixels_3d(
    grid_3d, requested_3d, artery_id, allowed_donors, protected_labels
): # 3D volume grid, requested 3D grid, artery id, allowed donors set, protected labels set
    """3D extension for borrowing tissue voxels across a volumetric grid."""
    updated = grid_3d.copy() # protect from overwrite
    blocked = np.zeros_like(grid_3d, dtype=bool) # initialize a block mask
    donor_counts = {} # donor counts dictionary

    # 1. Protection takes precedence over permission
    valid_donors = set(allowed_donors) - set(protected_labels) # the valid donors

    # 2. Identify voxels that are already the target artery
    is_already_artery = grid_3d == artery_id

    # 3. Borrowable mask: requested AND in valid donors AND NOT protected
    borrowable_mask = (
        requested_3d # requested
        & np.isin(grid_3d, list(valid_donors)) # in valid donors but not in protected labels
        & ~np.isin(grid_3d, list(protected_labels))
    )

    # 4. Blocked mask: requested, NOT borrowable, AND NOT already target artery
    blocked_mask = requested_3d & ~borrowable_mask & ~is_already_artery # blocked mask criteria
    blocked[blocked_mask] = True # changing the correct voxels to true for blocked

    # 5. Apply target replacement
    borrowed_labels = grid_3d[borrowable_mask] # which labels will be changed
    updated[borrowable_mask] = artery_id # changing the labels

    # 6. Tally donor voxel counts
    if borrowed_labels.size > 0: # if there are any changes, then proceeed
        unique_labels, counts = np.unique(borrowed_labels, return_counts=True) # deduplicate and sort
        for label, count in zip(unique_labels, counts): # zip
            donor_counts[int(label)] = int(count) # apply count

    return updated, donor_counts, blocked

# borrow pixels
def borrow_pixels_2d(grid, requested, artery_id, allowed_donors, protected_labels):  #np_array of integers, true/false binary grid of pixel requests, id to match, what can be borrowed, what is protected
    updated = grid.copy() #create copy
    blocked = np.zeros_like(grid, dtype=bool) # initial unblocked mask
    donor_counts = {} # empty dict

    # 1. Protection > permission
    valid_donors = set(allowed_donors) - set(protected_labels) # valid donors set

    # 2. Identify pixels that are already the artery
    is_already_artery = (grid == artery_id)

    # 3. Borrowable: requested AND valid_donors AND NOT protected_labels
    borrowable_mask = ( # this is the bool mask which says which can and cannot be changed into artery
        requested #must be requested
        & np.isin(grid, list(valid_donors)) # must be in valid donors
        & ~np.isin(grid, list(protected_labels)) # must not be in protected labels
    )

    # 4. Blocked pixels: requested, NOT borrowable, AND NOT already the target artery
    blocked_mask = requested & ~borrowable_mask & ~is_already_artery #creating the rule
    blocked[blocked_mask] = True # applying the rule

    # 5. Apply updates
    borrowed_labels = grid[borrowable_mask] #grid of the borrowable label ids
    updated[borrowable_mask] = artery_id #updated

    # 6. Tally donor counts for newly borrowed pixels only
    if borrowed_labels.size > 0:
        unique_labels, counts = np.unique(borrowed_labels, return_counts=True) # 2 arrays: unique tissues and their counts after deduplicating and sorting
        for label, count in zip(unique_labels, counts): # populate donor counts dictionary after comparing the zip of the two arrays
            donor_counts[int(label)] = int(count)

    return updated, donor_counts, blocked

TISSUE_PROPERTIES = {
    OUTSIDE: {"resistance": 999, "borrowable": False, "cohesion": 0.0},
    SKIN:    {"resistance":  50, "borrowable": False, "cohesion": 0.8},
    FAT:     {"resistance":   2, "borrowable": True,  "cohesion": 0.2},
    MUSCLE:  {"resistance":   8, "borrowable": True,  "cohesion": 0.5},
    BONE:    {"resistance": 999, "borrowable": False, "cohesion": 1.0},
    ARTERY:  {"resistance":   0, "borrowable": False, "cohesion": 0.0},
}

# --- 3. HELPER TO CONVERT GRID TO PROPERTY MAPS ---
def create_property_maps(labels_grid, property_dict):
    """Generates continuous numeric maps for resistance and borrow permissions."""
    resistance_map = np.zeros(labels_grid.shape, dtype=float)
    borrowable_map = np.zeros(labels_grid.shape, dtype=bool)

    for tissue_id, props in property_dict.items():
        mask = (labels_grid == tissue_id)
        resistance_map[mask] = props["resistance"]
        borrowable_map[mask] = props["borrowable"]

    return resistance_map, borrowable_map

# --- 4. SCORE-BASED EXPANSION FUNCTION ---
def score_based_borrow_2d(labels, push_force, property_dict, artery_id=ARTERY):
    """
    Evaluates candidate expansion pixels against accumulated push_force and tissue costs.
    - Each pixel's required energy = Resistance / Permission_Factor
    - Voxels are borrowed if push_force >= Resistance
    """
    updated_labels = labels.copy()
    blocked_mask = np.zeros_like(labels, dtype=bool)
    donor_counts = {}

    # Extract current artery and candidate 1-pixel outer boundary (8-connected)
    artery_mask = (labels == artery_id)
    dilated_boundary = binary_dilation(artery_mask, structure=np.ones((3, 3)))
    candidate_ring = dilated_boundary & ~artery_mask

    # Map physical properties onto the spatial grid
    resistance_map, borrowable_map = create_property_maps(labels, property_dict)

    # Evaluate candidates
    for y, x in zip(*np.where(candidate_ring)):
        cost = resistance_map[y, x]
        can_borrow = borrowable_map[y, x]

        # Check force threshold against localized resistance
        if can_borrow and push_force >= cost:
            # Overcame resistance: convert voxel to Artery
            original_tissue = labels[y, x]
            updated_labels[y, x] = artery_id
            donor_counts[original_tissue] = donor_counts.get(original_tissue, 0) + 1
        else:
            # Blocked either due to protection (can_borrow=False) or insufficient push force
            blocked_mask[y, x] = True

    return updated_labels, donor_counts, blocked_mask

def vti_xml_without_appended(vti_text):
    """Return parseable XML metadata without the binary/base64 payload."""
    return vti_text.split("<AppendedData", 1)[0] + "</VTKFile>"


def vti_appended_payload_b64(vti_text):
    match = re.search(r"<AppendedData[^>]*>\s*_(.*?)\s*</AppendedData>", vti_text, re.S)
    if match is None:
        raise ValueError("The VTI has no appended base64 payload.")
    return "".join(match.group(1).split())


def vti_decode_vtk_base64(payload):
    """Decode VTK's sometimes-concatenated base64 chunks."""
    payload = "".join(payload.split())
    if "=" not in payload:
        return base64.b64decode(payload)
    chunks, start, i = [], 0, 0
    while i < len(payload):
        if payload[i] != "=":
            i += 1
            continue
        j = i
        while j < len(payload) and payload[j] == "=":
            j += 1
        if payload[start:j]:
            chunks.append(base64.b64decode(payload[start:j]))
        start, i = j, j
    if payload[start:]:
        chunks.append(base64.b64decode(payload[start:]))
    return b"".join(chunks)


def vti_read_activity_zyx(vti_path):
    """Read the scalar CellData array in this VTI without installing VTK."""
    vti_text = Path(vti_path).read_text(errors="ignore")
    root = ET.fromstring(vti_xml_without_appended(vti_text))
    if root.attrib.get("type") != "ImageData":
        raise ValueError("Expected a VTK ImageData (.vti) file.")
    image = root.find("ImageData")
    piece = image.find("Piece") if image is not None else None
    if piece is None:
        raise ValueError("VTI is missing ImageData/Piece metadata.")
    data_array = piece.find("./CellData/DataArray[@Name='activity']")
    if data_array is None:
        data_array = piece.find("./CellData/DataArray")
    if data_array is None:
        raise ValueError("No scalar CellData array was found in this VTI.")
    if data_array.attrib.get("format") != "appended":
        raise ValueError("This compact reader currently expects appended VTI data.")

    byte_order = root.attrib.get("byte_order", "LittleEndian")
    endian = "<" if byte_order == "LittleEndian" else ">"
    vtk_types = {"Float32": "f4", "Float64": "f8", "Int16": "i2", "UInt16": "u2",
                 "Int32": "i4", "UInt32": "u4", "Int64": "i8", "UInt64": "u8"}
    data_type = data_array.attrib.get("type", "Float32")
    if data_type not in vtk_types:
        raise ValueError(f"Unsupported VTI scalar type: {data_type}")
    data_dtype = np.dtype(endian + vtk_types[data_type])
    header_type = root.attrib.get("header_type", "UInt32")
    header_dtype = np.dtype(endian + ("u8" if header_type == "UInt64" else "u4"))

    extent = tuple(map(int, piece.attrib.get("Extent", image.attrib["WholeExtent"]).split()))
    x0, x1, y0, y1, z0, z1 = extent
    # CellData has one fewer item than points along each direction.

    expected_count = nx * ny * nz

    payload_b64 = vti_appended_payload_b64(vti_text)
    offset = int(data_array.attrib["offset"])

    def vti_uncompress_at(raw_bytes, byte_offset):
        """Unpack VTK zlib blocks at a byte offset into an already-decoded stream."""
        header = np.frombuffer(raw_bytes, dtype=header_dtype, count=3, offset=byte_offset).astype(np.int64)
        n_blocks = int(header[0])
        if n_blocks <= 0 or n_blocks > 1_000_000:
            raise ValueError("Invalid VTI compressed-block count.")
        word = header_dtype.itemsize
        size_offset = byte_offset + 3 * word
        compressed_sizes = np.frombuffer(raw_bytes, dtype=header_dtype, count=n_blocks, offset=size_offset)
        cursor = size_offset + n_blocks * word
        chunks = []
        for size in compressed_sizes:
            size = int(size)
            block = raw_bytes[cursor:cursor + size]
            if len(block) != size:
                raise ValueError("VTI compressed block is truncated.")
            cursor += size
            chunks.append(zlib.decompress(block))
        return b"".join(chunks)

    # VTK writers disagree about whether appended offsets count decoded bytes or
    # base64 characters.  Try the normal decoded-byte interpretation first,
    # then the encoded-character interpretation used by this arm.vti source.
    try:
        unpacked = vti_uncompress_at(vti_decode_vtk_base64(payload_b64), offset)
    except Exception:
        if offset < 0 or offset >= len(payload_b64):
            raise ValueError("VTI appended-data offset is outside the base64 payload.")
        unpacked = vti_uncompress_at(vti_decode_vtk_base64(payload_b64[offset:]), 0)
    flat = np.frombuffer(unpacked, dtype=data_dtype, count=expected_count).copy()
    if flat.size != expected_count:
        raise ValueError(f"Expected {expected_count} values but read {flat.size} from the VTI.")

    spacing_xyz = tuple(map(float, image.attrib.get("Spacing", "1 1 1").split()))
    origin_xyz = tuple(map(float, image.attrib.get("Origin", "0 0 0").split()))
    return flat.reshape((nz, ny, nx)), spacing_xyz[::-1], origin_xyz[::-1]


def vti_resolve_input_path():
    """Locate a local VTI, accept a Colab upload, or fetch the tracked file."""
    candidates = []
    if VTI_LOCAL_PATH:
        candidates.append(Path(VTI_LOCAL_PATH))
    candidates.extend([Path("/content/arm.vti"), Path("arm.vti"), Path("data/arm.vti")])
    if "REFERENCE_DIR" in globals():
        candidates.append(Path(REFERENCE_DIR) / "data" / "arm.vti")
    for candidate in candidates:
        if candidate.is_file():
            return candidate
    if VTI_LOAD_MODE.lower() == "upload":
        try:
            from google.colab import files
        except ImportError as exc:
            raise FileNotFoundError("Upload mode is available only in Google Colab.") from exc
        uploaded = files.upload()
        if not uploaded:
            raise FileNotFoundError("No VTI was uploaded.")
        uploaded_name = next(iter(uploaded))
        return Path(uploaded_name)
    destination = Path("/content/arm.vti") if Path("/content").is_dir() else Path("arm.vti")
    try:
        print(f"Downloading {VTI_GITHUB_URL}")
        urllib.request.urlretrieve(VTI_GITHUB_URL, destination)
    except Exception as exc:
        raise FileNotFoundError(
            "Could not find arm.vti locally or download it. Set VTI_LOCAL_PATH, "
            "use VTI_LOAD_MODE='upload' in Colab, or clone the FakeCT repo."
        ) from exc
    return destination


# -----------------------------------------------------------------------------
# 3) SCALAR -> EDITABLE CATEGORICAL BANDS (kept separate from intensity data)
# -----------------------------------------------------------------------------
def vti_make_teaching_bands(intensity):
    """Keep exact zero as background; assign all other values to editable bands."""
    edges = np.asarray(VTI_BAND_EDGES, dtype=float)
    if len(edges) != 8 or not np.all(np.diff(edges[1:-1]) > 0):
        raise ValueError("VTI_BAND_EDGES must be (-inf, six increasing cuts, inf).")
    labels = np.zeros(intensity.shape, dtype=np.uint8)
    non_background = ~np.isclose(intensity, VTI_BACKGROUND_VALUE, atol=VTI_BACKGROUND_ATOL)
    # digitize returns 0..6 for the seven non-background intervals; add 1.
    labels[non_background] = np.digitize(intensity[non_background], edges[1:-1], right=False) + 1
    return labels


def vti_make_sphere(shape, center_zyx, radius_mm, spacing_zyx):
    z, y, x = np.ogrid[:shape[0], :shape[1], :shape[2]]
    dz = (z - center_zyx[0]) * spacing_zyx[0]
    dy = (y - center_zyx[1]) * spacing_zyx[1]
    dx = (x - center_zyx[2]) * spacing_zyx[2]
    return dz * dz + dy * dy + dx * dx <= radius_mm * radius_mm


def vti_find_illustrative_seed(labels, donor_label, spacing_zyx, radius_mm, border_mm):
    """Pick a large interior region of the selected scalar band; it is not anatomy detection."""
    interior_distance = distance_transform_edt(labels == donor_label, sampling=spacing_zyx)
    margin = np.ceil(border_mm / np.asarray(spacing_zyx)).astype(int)
    eligible = np.ones(labels.shape, dtype=bool)
    for axis, width in enumerate(margin):
        if width > 0:
            low = [slice(None)] * 3
            high = [slice(None)] * 3
            low[axis] = slice(0, width)
            high[axis] = slice(labels.shape[axis] - width, labels.shape[axis])
            eligible[tuple(low)] = False
            eligible[tuple(high)] = False
    candidate_distance = np.where(eligible, interior_distance, -1.0)
    center = tuple(map(int, np.unravel_index(np.argmax(candidate_distance), labels.shape)))
    if candidate_distance[center] < radius_mm:
        raise ValueError(
            "No sufficiently interior illustrative seed was found. Choose a different "
            "VTI_TARGET_SOURCE_LABEL, reduce VTI_SEED_RADIUS_MM, or set VTI_SEED_CENTER_ZYX."
        )
    return center


# -----------------------------------------------------------------------------
# 4) SHAPE-CONSTRAINED SIX-NEIGHBOR DILATION + LABEL-AWARE BORROWING
# -----------------------------------------------------------------------------
def vti_dilate6(mask):
    """One 6-connected bitwise dilation: each voxel sees face-sharing neighbors only."""
    expanded = mask.copy()
    expanded[1:, :, :] |= mask[:-1, :, :]
    expanded[:-1, :, :] |= mask[1:, :, :]
    expanded[:, 1:, :] |= mask[:, :-1, :]
    expanded[:, :-1, :] |= mask[:, 1:, :]
    expanded[:, :, 1:] |= mask[:, :, :-1]
    expanded[:, :, :-1] |= mask[:, :, 1:]
    return expanded


def vti_shape_domain(shape, center_zyx, target_radius_mm, spacing_zyx):
    """Spatial request domain for a non-symmetric lesion-like dilation experiment."""
    shape_type = VTI_SHAPE_TYPE.lower()
    if shape_type not in {"sphere", "ellipsoid", "focal", "tapered"}:
        raise ValueError("VTI_SHAPE_TYPE must be sphere, ellipsoid, focal, or tapered.")
    asymmetry = float(np.clip(VTI_ASYMMETRY, 0.0, 0.9))
    axis_scale = np.asarray(VTI_AXIS_SCALE_ZYX, dtype=float)
    if axis_scale.shape != (3,) or np.any(axis_scale <= 0):
        raise ValueError("VTI_AXIS_SCALE_ZYX must contain three positive (z, y, x) values.")
    direction = np.asarray(VTI_ASYMMETRY_DIRECTION_ZYX, dtype=float)
    if direction.shape != (3,) or np.linalg.norm(direction) == 0:
        raise ValueError("VTI_ASYMMETRY_DIRECTION_ZYX must be a nonzero (z, y, x) vector.")
    direction /= np.linalg.norm(direction)

    zz, yy, xx = np.ogrid[:shape[0], :shape[1], :shape[2]]
    relative = np.broadcast_arrays(
        (zz - center_zyx[0]) * spacing_zyx[0],
        (yy - center_zyx[1]) * spacing_zyx[1],
        (xx - center_zyx[2]) * spacing_zyx[2],
    )
    axial = relative[0] * direction[0] + relative[1] * direction[1] + relative[2] * direction[2]
    squared_norm = relative[0] ** 2 + relative[1] ** 2 + relative[2] ** 2
    radial = np.sqrt(np.maximum(0.0, squared_norm - axial ** 2))

    if shape_type == "sphere":
        shift = asymmetry * (target_radius_mm - VTI_SEED_RADIUS_MM)
        shifted_sq = (relative[0] - shift * direction[0]) ** 2
        shifted_sq += (relative[1] - shift * direction[1]) ** 2
        shifted_sq += (relative[2] - shift * direction[2]) ** 2
        return shifted_sq <= target_radius_mm ** 2

    if shape_type == "ellipsoid":
        shift = asymmetry * (target_radius_mm - VTI_SEED_RADIUS_MM)
        scaled_sq = ((relative[0] - shift * direction[0]) / (target_radius_mm * axis_scale[0])) ** 2
        scaled_sq += ((relative[1] - shift * direction[1]) / (target_radius_mm * axis_scale[1])) ** 2
        scaled_sq += ((relative[2] - shift * direction[2]) / (target_radius_mm * axis_scale[2])) ** 2
        return scaled_sq <= 1.0

    # "focal" and "tapered" use the chosen direction as an axial lesion axis.
    # Convert the editable axis scales into an axial and a transverse scale for
    # an arbitrarily directed focal/tapered lesion.  Thus changing an element
    # of VTI_AXIS_SCALE_ZYX genuinely changes the requested 3-D geometry.
    axial_scale = float(np.linalg.norm(axis_scale * direction))
    transverse_scale = float(np.sqrt(max((np.sum(axis_scale ** 2) - axial_scale ** 2) / 2.0, 1e-9)))
    axis_length = target_radius_mm * axial_scale
    transverse_radius = target_radius_mm * transverse_scale
    if shape_type == "focal":
        shifted_axial = axial - asymmetry * (target_radius_mm - VTI_SEED_RADIUS_MM)
        axial_fraction = shifted_axial / axis_length
        local_radius = transverse_radius * (
            0.35 + 0.65 * np.exp(-VTI_FOCAL_SHARPNESS * axial_fraction ** 2)
        )
        return (np.abs(shifted_axial) <= axis_length) & (radial <= local_radius)

    # Tapered: asymmetric length and a steadily narrowing far end.
    negative_extent = axis_length * (1.0 - asymmetry)
    positive_extent = axis_length * (1.0 + asymmetry)
    axial_position = (axial + negative_extent) / (negative_extent + positive_extent)
    local_radius = transverse_radius * (
        1.0 - (1.0 - VTI_TAPER_END_RADIUS_FRACTION) * np.clip(axial_position, 0.0, 1.0)
    )
    return (
        (axial >= -negative_extent)
        & (axial <= positive_extent)
        & (radial <= local_radius)
    )


def vti_borrow_round(labels_before_round, requested, artery_label):
    """Reassign allowed donor bands simultaneously and retain a transparent audit."""
    eligible = np.zeros(labels_before_round.shape, dtype=bool)
    for label, policy in VTI_BAND_POLICY.items():
        eligible |= (
            (labels_before_round == label)
            & bool(policy["borrowable"])
            & (float(policy["resistance"]) <= VTI_BORROW_FORCE)
        )
    claimed = requested & eligible & (labels_before_round != artery_label)
    blocked = requested & ~claimed
    labels_after_round = labels_before_round.copy()
    donor_counts = {
        int(label): int(np.count_nonzero(claimed & (labels_before_round == label)))
        for label in sorted(VTI_BAND_POLICY)
        if np.any(claimed & (labels_before_round == label))
    }
    labels_after_round[claimed] = artery_label
    return labels_after_round, claimed, blocked, donor_counts

def vti_native_index(sampled_index):
    """Map a displayed sampled index back to its source-VTI index."""
    return int(vti_sample_offset + int(sampled_index) * int(VTI_STRIDE))


def vti_preview_z_indices():
    """Use user-selected slices or a useful overview/local survey automatically."""
    nz = vti_labels_original.shape[0]
    if VTI_PREVIEW_Z_INDICES is not None:
        indices = sorted({int(value) for value in VTI_PREVIEW_Z_INDICES})
        bad = [value for value in indices if value < 0 or value >= nz]
        if bad:
            raise ValueError(f"VTI_PREVIEW_Z_INDICES contains out-of-bounds z indices: {bad}")
        return indices
    if vti_manual_roi_center_zyx is None:
        # First pass: survey the full volume to locate a plausible z index.
        return np.unique(np.rint(np.linspace(0, nz - 1, VTI_PREVIEW_NUM_Z_SLICES)).astype(int)).tolist()
    # Subsequent passes: inspect a local stack spanning the manual ROI.
    radius_z_vox = VTI_ROI_RADIUS_MM / vti_spacing_zyx[0]
    low = max(0, int(np.floor(vti_manual_roi_center_zyx[0] - 1.25 * radius_z_vox)))
    high = min(nz - 1, int(np.ceil(vti_manual_roi_center_zyx[0] + 1.25 * radius_z_vox)))
    return np.unique(np.rint(np.linspace(low, high, VTI_PREVIEW_NUM_Z_SLICES)).astype(int)).tolist()


def vti_add_roi_overlay(ax, z_index):
    """Draw the true plane/sphere intersection and crosshair in an axial panel."""
    if not VTI_SHOW_ROI_OVERLAY or vti_manual_roi_center_zyx is None:
        return None
    center_z, center_i, center_j = vti_manual_roi_center_zyx
    delta_z_mm = (z_index - center_z) * vti_spacing_zyx[0]
    if abs(delta_z_mm) <= VTI_ROI_RADIUS_MM:
        radius_in_plane_mm = np.sqrt(VTI_ROI_RADIUS_MM ** 2 - delta_z_mm ** 2)
        ellipse = Ellipse(
            (center_j, center_i),
            width=2.0 * radius_in_plane_mm / vti_spacing_zyx[2],
            height=2.0 * radius_in_plane_mm / vti_spacing_zyx[1],
            fill=False,
            edgecolor="#00d9ff",
            linewidth=2.0,
            linestyle="--",
        )
        ax.add_patch(ellipse)
        ax.plot(center_j, center_i, marker="+", markersize=10, markeredgewidth=2.0, color="#00d9ff")
        return radius_in_plane_mm
    return None

def vti_plane(array, plane):
    if plane == "axial":
        return array[vti_seed_center_zyx[0], :, :]
    if plane == "coronal":
        return array[:, vti_seed_center_zyx[1], :]
    return array[:, :, vti_seed_center_zyx[2]]


def vti_claim_block_overlay(plane):
    base = vti_plane(vti_labels_seeded, plane) > 0
    rgb = np.repeat((0.20 + 0.35 * base)[..., None], 3, axis=-1)
    claimed = vti_plane(vti_claimed_mask, plane)
    blocked = vti_plane(vti_blocked_mask, plane)
    rgb[claimed] = (0.10, 0.45, 0.90)  # blue = borrowed/reassigned
    rgb[blocked] = (0.88, 0.20, 0.16)  # red = requested but protected/too resistant
    return rgb
